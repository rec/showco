def authorize_url(config_path: Path, values: dict[str, object]) -> int:
    stream = config.table_value(values, 'stream')
    client_id = require_value(stream, 'client_id', config_path)
    redirect_uri = require_value(stream, 'redirect_uri', config_path)
    scopes = require_value(stream, 'scopes', config_path)
    state = secrets.token_urlsafe(24)
    write_toml_value(config_path, 'stream', 'state', state)
    params = {
        'response_type': 'code',
        'client_id': client_id,
        'redirect_uri': redirect_uri,
        'scope': scopes,
        'state': state,
        'force_verify': 'true',
    }
    url = STREAM_AUTHORIZE_URL + '?' + parse.urlencode(params)
    webbrowser.open(url)
    print(
        'After approving it, copy the full localhost callback URL from the browser\n'
        'address bar and paste it into stream.callback_url in secrets.toml.'
    )
    return 0

def exchange_code(config_path: Path, secrets_path: Path) -> int:
    env = config.load_values(config_path, secrets_path)
    stream = config.table_value(env, 'stream')
    client_id = require_value(stream, 'client_id')
    client_secret = require_value(stream, 'client_secret')
    redirect_uri = require_value(stream, 'redirect_uri')
    callback = require_value(stream, 'callback_url')
    code = callback_code(callback, require_value(stream, 'state'), redirect_uri)
    config_dir = Path.home() / '.config/streamo'
    config_dir.mkdir(mode=0o700, parents=True, exist_ok=True)
    config_dir.chmod(0o700)
    response_file = config_dir / 'oauth-response.json'
    data = parse.urlencode(
        {
            'client_id': client_id,
            'client_secret': client_secret,
            'code': code,
            'grant_type': 'authorization_code',
            'redirect_uri': redirect_uri,
        }
    ).encode()
    http_request = request.Request(STREAM_TOKEN_URL, data=data, method='POST')
    http_response = request_http(http_request)
    write_private(response_file, http_response.text + '\n')
    try:
        response = json.loads(http_response.text)
    except json.JSONDecodeError:
        print(
            'Stream returned non-JSON token response. '
            f'Response saved to {response_file}'
        )
        return 1
    if http_response.status < 200 or http_response.status >= 300:
        print(
            f'Stream token request failed with HTTP {http_response.status}. '
            f'Response saved to {response_file}'
        )
        print(json.dumps(response, indent=2))
        return 1

    if 'access_token' not in response:
        message = (
            f'Stream did not return an access token. Response saved to {response_file}'
        )
        print(message)
        print(json.dumps(response, indent=2))
        return 1

    access_token = response['access_token']
    if not isinstance(access_token, str):
        raise ValueError('Stream returned an invalid access token')
    with atomic_output(secrets_path) as temporary:
        if secrets_path.exists():
            settings.write_text_atomically(temporary, secrets_path.read_text())
        write_toml_value(temporary, 'stream', 'access_token', access_token)
    write_private(config_dir / 'oauth-token', access_token + '\n')
    if refresh_token := response.get('refresh_token'):
        write_private(config_dir / 'refresh-token', refresh_token + '\n')

    print(f'Saved full response to {response_file}')
    print(f'Saved access token to {config_dir / "oauth-token"}')
    if response.get('refresh_token'):
        print(f'Saved refresh token to {config_dir / "refresh-token"}')
    print()
    print('Run showco streamo validate-token next.')
    return 0

def validate_token(values: dict[str, object]) -> int:
    stream = config.table_value(values, 'stream')
    token = require_value(stream, 'access_token')
    http_request = request.Request(
        STREAM_VALIDATE_URL,
        headers={'Authorization': f'OAuth {token}'},
    )
    http_response = request_http(http_request)
    if http_response.status < 200 or http_response.status >= 300:
        sys.exit(f'Stream token validation failed with HTTP {http_response.status}.')
    try:
        response = json.loads(http_response.text)
    except json.JSONDecodeError:
        sys.exit('Stream returned non-JSON token validation response.')
    print(json.dumps(response, indent=4))
    return 0

def callback_code(value: str, state: str, redirect_uri: str) -> str:
    url = parse.urlsplit(value)
    expected = parse.urlsplit(redirect_uri)
    if url.scheme not in {'http', 'https'} or (url.scheme, url.netloc, url.path) != (
        expected.scheme,
        expected.netloc,
        expected.path,
    ):
        raise ValueError('Paste the full OAuth callback URL matching redirect_uri')
    params = parse.parse_qs(url.query, keep_blank_values=True)
    if params.get('state') != [state] or not state:
        raise ValueError(
            'OAuth callback state does not match the pending authorization'
        )
    if 'error' in params:
        raise ValueError('OAuth authorization was denied: ' + params['error'][0])
    codes = params.get('code', [])
    if len(codes) != 1 or not codes[0]:
        raise ValueError('OAuth callback must contain exactly one nonempty code')
    return codes[0]

def request_http(http_request: request.Request) -> HttpResponse:
    try:
        with request.urlopen(http_request, timeout=10) as response:
            return HttpResponse(status=response.status, text=response.read().decode())
    except error.HTTPError as e:
        return HttpResponse(status=e.code, text=e.read().decode())
    except OSError as failure:
        sys.exit(f'OAuth HTTP request failed: {failure}')

def write_toml_value(path: Path, section: str, name: str, value: str) -> None:
    text = path.read_text() if path.exists() else ''
    tomllib.loads(text)
    lines = iter(text.splitlines(keepends=True))
    replacement = f'{name} = {json.dumps(value)}\n'
    header = tomllib.loads(f'[{section}]')
    result: list[str] = []
    in_section = False
    found_section = False
    changed = False
    for line in lines:
        statement = line
        while True:
            try:
                parsed = tomllib.loads(statement)
                break
            except tomllib.TOMLDecodeError:
                statement += next(lines)
        if statement.lstrip().startswith('['):
            if in_section and not changed:
                result.append(replacement)
                changed = True
            in_section = parsed == header
            found_section |= in_section
        elif in_section and name in parsed:
            result.append(replacement)
            changed = True
            continue
        result.append(statement if statement.endswith('\n') else statement + '\n')
    if not changed:
        if not found_section:
            result.append(f'\n[{section}]\n')
        result.append(replacement)
    updated = ''.join(result)
    try:
        tomllib.loads(updated)
    except tomllib.TOMLDecodeError:
        raise ValueError(
            f'Cannot update {section}.{name} in this TOML layout; file unchanged'
        ) from None
    settings.write_text_atomically(path, updated)

def require_value(env: dict[str, object], name: str, path: Path | None = None) -> str:
    value = env.get(name, '')
    if not isinstance(value, str):
        sys.exit(f'Set {name} to a string.')
    if value and value != 'TODO':
        return value
    if path is None:
        sys.exit(f'Set {name} first.')
    sys.exit(f'Edit {path} and set {name} first.')
